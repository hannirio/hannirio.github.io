### [79. Word Search](https://leetcode.com/problems/word-search/description/)
在面试的时候碰到这道题了，思路确实很简单，但就是控制DFS和回溯的时候，细节挺多的。我觉得这道题的练习意义非常重大，相当于把DFS和回溯全部练了一遍了。标准套模板的代码如下。

```cpp

class Solution {
public:
    // 偏移数组
    vector<int> dx = {0, -1, 0, 1};
    vector<int> dy = {-1, 0, 1, 0};
    // 矩阵规模
    int m, n;
    // 访问数组
    vector<vector<bool>> visited;
    // 越界判断: 出边界 or 已访问(放在边界后/防止越界)
    bool isValid(int x, int y, int m, int n){
        if(x < 0 || x>= m || y < 0 || y >= n || visited[x][y]) return false;
        return true;
    }
    // 回溯终止判断：找到路径
    bool find;

    // DFS回溯
    void dfs(int x, int y, int m, int n, int idx, vector<vector<char>>& board, string word){
        if(idx == word.size()){  // 找到立刻返回，防止越界
            find = true;
            return;
        }
        for(int i = 0; i < 4; i++){
            int nx = x + dx[i];
            int ny = y + dy[i];
            if(isValid(nx, ny, m, n) && board[nx][ny] == word[idx]){
                visited[nx][ny] = true;
                dfs(nx, ny, m, n, idx + 1, board, word);
                if(find) return;
                visited[nx][ny] = false;   // 回溯！！！！！
            }
        }
    }

    bool exist(vector<vector<char>>& board, string word) {
        // 初始化
        m = board.size(), n = board[0].size();
        find = false;
        visited = vector<vector<bool>>(m, vector<bool>(n, false));
        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if(board[i][j] == word[0]){
                    visited[i][j] = true;
                    dfs(i, j, m, n, 1, board, word);
                    visited[i][j] = false;
                }
            }
        }
        return find;
    }
};
```
精简之后的做法也很不错,直接对所有点进行一个DFS,优化掉原有的visited数组，直接进行一个原地修改
```cpp

class Solution {
private:
    bool dfs(int x, int y, int idx, vector<vector<char>>& board, const string& word){
        // 越界判断
        if(x < 0|| x >= board.size() || y < 0 || y >= board[0].size() || board[x][y] != word[idx]) return false;
        // 递归终点
        if(idx == word.size() - 1) return true;
        // 原地修改
        char tmp = board[x][y];
        board[x][y] = '#';
        bool found = dfs(x, y - 1, idx + 1, board, word) ||
                     dfs(x - 1, y, idx + 1, board, word) ||
                     dfs(x, y + 1, idx + 1, board, word) ||
                     dfs(x + 1, y, idx + 1, board, word);
        // 回溯返回
        board[x][y] = tmp;
        return found;
    }
public:
    bool exist(vector<vector<char>>& board, string word) {
        for(int i = 0; i < board.size(); i++){
            for(int j = 0; j < board[0].size(); j++){
                if(dfs(i, j, 0, board, word)) return true;
            }
        }
        return false;
    }
};
```

### [605. Can Place Flowers](https://leetcode.com/problems/can-place-flowers/description/)
这道题的思路直接贪心就过了，算是非常经典的一个贪心算法题目了。比较麻烦的是这个if判断逻辑。


```cpp

class Solution {
public:
    bool canPlaceFlowers(vector<int>& flowerbed, int n) {
        int cnt = 0;
        for(int i = 0; i < flowerbed.size(); i++){
            // 判断当前位置及左右相邻位置是否为空
            if(flowerbed[i] == 0 && (i == 0 || flowerbed[i - 1] == 0) && (i == flowerbed.size() - 1 || flowerbed[i + 1] == 0)){
                cnt++;
                flowerbed[i] = 1;
                if(cnt >= n) return true;
            }
        }
        return cnt >= n;
    }
};
```

### [97. Interleaving String](https://leetcode.com/problems/interleaving-string/description/)
非常经典的一个2维DP，建出DP数组之后就比较容易了
```cpp

class Solution {
public:
    bool isInterleave(string s1, string s2, string s3) {
        int n = s1.size(), m = s2.size();
        if(n + m < s3.size()) return false;
        vector<vector<bool>> dp(n + 1, vector<bool>(m + 1, false));
        dp[0][0] = true;
        // dp[i][j] 表示s1的前i-1的子串➕s2的前j-1的子串是否能够构成s3
        for(int i = 0; i <= n; i++){
            for(int j = 0; j <= m; j++){
                int k = i + j - 1;  // 比对s3的位置
                if(i > 0){
                    dp[i][j] = dp[i][j] || (dp[i - 1][j] && s1[i - 1] == s3[k]);
                }
                if(j > 0){
                    dp[i][j] = dp[i][j] || (dp[i][j - 1] && s2[j - 1] == s3[k]);
                }
            }
        }
        return dp[n][m];
    }
};
```

### [1712. Ways to Split Array Into Three Subarrays](https://leetcode.com/problems/ways-to-split-array-into-three-subarrays/description/)
思路：利用前缀和 + 二分搜索 （ 总共砍2刀，枚举固定第一刀的位置然后找出可能的第二刀的位置）
```cpp

class Solution {
public:
    int waysToSplit(vector<int>& nums) {
        int n = nums.size();
        vector<int> pref(n);
        pref[0] = nums[0];
        for(int i = 1; i < n; i++) pref[i] = pref[i - 1] + nums[i];
        long long ans = 0;
        int Mod = 1e9 + 7;
        // 枚举第一刀的位置
        for(int i = 0; i < n - 2; i++){
            int target_min = 2 * pref[i];
            int target_max = (pref[n - 1] + pref[i]) / 2;
            if(target_min > target_max) continue;
            auto l = lower_bound(pref.begin() + i + 1, pref.end() - 1, target_min);
            auto r = upper_bound(pref.begin() + i + 1, pref.end() - 1, target_max);
            if(l < r) ans = (ans + (r - l)) % Mod;
        }
        return ans;
    }
};
```

### [8. String to Integer (atoi)](https://leetcode.com/problems/string-to-integer-atoi/description/)

遍历字符串判断越界，综合来说就是模拟题目


```cpp

class Solution {
public:
    int myAtoi(string s) {
        int i = 0;
        int n = s.size();
        // 处理空白格
        while(i < n && s[i] == ' ') i++;
        if(i == n) return 0;   // 全为空格的情况

        // 处理符号
        int sgn = 1;
        if(s[i] == '-'){
            sgn = -1;
            i++;
        }
        else if(s[i] == '+') i++;

        // 处理数字并判断越界
        int res = 0;
        while(i < n && s[i] >= '0' && s[i] <= '9'){
            int digit = s[i] - '0';
            if(res > INT_MAX / 10 || (res == INT_MAX / 10 && digit > 7)){
                return sgn == 1 ? INT_MAX : INT_MIN;
            }
            res = res * 10 + digit;
            i++;
        }
        return res * sgn;
    }
};
```

### [395. Longest Substring with At Least K Repeating Characters](https://leetcode.com/problems/longest-substring-with-at-least-k-repeating-characters/description/)
思路：使用分治法然后递归解决

```cpp

class Solution {
public:
    int longestSubstring(string s, int k) {
        if(s.size() == 0 || k > s.size()) return 0;
        unordered_map<char, int> mp;
        for(char c : s) mp[c]++;
        for(int i = 0; i < s.size(); i++){
            if(mp[s[i]] < k){
                string left_part = s.substr(0, i);
                int j = i + 1;
                while(j < s.size() && mp[s[j]] < k){
                    j++;
                }
                string right_part = s.substr(j);
                return max(longestSubstring(left_part, k), longestSubstring(right_part, k));
            }
        } 
        return s.size();
    }
};
```

### [525. Contiguous Array](https://leetcode.com/problems/contiguous-array/description/)
方法：使用前缀和➕哈希
```cpp

class Solution {
public:
    int findMaxLength(vector<int>& nums) {
        // sum : idx
        unordered_map<int, int> mp;
        int max_len = 0;
        int sum = 0;
        mp[0] = -1;   // 前缀和为0，出现在索引-1
        for(int i = 0; i < nums.size(); i++){
            sum += (nums[i] == 1 ? 1 : -1);  // 0当作-1，1当作1
            if(mp.count(sum)){
                max_len = max(max_len, i - mp[sum]);
            }
            else{
                mp[sum] = i;
            }
        }
        return max_len;
    }
};
```

### [46. Permutations](https://leetcode.com/problems/permutations/description/)
最经典的回溯板子题目，要把所有分支都遍历一遍，与下面一题有点区别。

``` cpp

class Solution {
public:
    vector<vector<int>> ans;
    void rec(vector<int>& nums, vector<bool>& visited, vector<int>& tmp){
        if(tmp.size() == nums.size()) ans.push_back(tmp);
        // 遍历所有分支
        for(int i = 0; i < nums.size(); i++){
            if(visited[i]) continue;
            visited[i] = true;
            tmp.push_back(nums[i]);
            rec(nums, visited, tmp);
            tmp.pop_back();      // 回溯阶段
            visited[i] = false;
        }
    }
    vector<vector<int>> permute(vector<int>& nums) {
        ans.clear();
        int n = nums.size();
        vector<bool> visited(n, false);
        vector<int> tmp;
        rec(nums, visited, tmp);
        return ans;
    }
}
```

### [78. Subsets](https://leetcode.com/problems/subsets/description/)
也是回溯的经典题目，根据idx来控制回溯的范围，与上题有点区别
```cpp

class Solution {
public:
    vector<vector<int>> ans;
    void rec(vector<int>& nums, vector<int>& tmp, int idx){
        if(tmp.size() <= nums.size()) ans.push_back(tmp);
        for(int i = idx; i < nums.size(); i++){
            tmp.push_back(nums[i]);
            // 控制范围剪枝
            rec(nums, tmp, i + 1);
            tmp.pop_back();
        }
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        ans.clear();
        vector<int> tmp;
        rec(nums, tmp, 0);
        return ans;
    }
};
```

### [146. LRU Cache](https://leetcode.com/problems/lru-cache/description/)
这道题应该是面试里面最最经典的题目了，不想去记双向链表的内置函数，那么就用数组来模拟双向链表来处理节点。
```cpp

class LRUCache {
private:
    int cap;
    int size;
    // 分别模拟虚拟头节点,尾节点
    int head = 10001;
    int tail = 10002;
    // 直接用连续数组构建数据结构，索引就是 Key
    int val[10005];
    int prev[10005];
    int nxt[10005];
    bool exist[10005];

    void remove(int key){
        int p = prev[key];
        int n = nxt[key];
        nxt[p] = n;
        prev[n] = p;
    }
    // 将节点插到头部之后
    void add_to_head(int key){
        int n = nxt[head];
        nxt[head] = key;
        prev[key] = head;
        nxt[key] = n;
        prev[n] = key;
    }


public:
    LRUCache(int capacity) {
        cap = capacity;
        size = 0;
        // 初始化首尾连接
        nxt[head] = tail;
        prev[tail] = head;
        // 初始化存在标记
        memset(exist, 0, sizeof(exist));
    }
    
    int get(int key) {
        if(!exist[key]) return -1;
        remove(key);
        add_to_head(key);
        return val[key];
    }
    
    void put(int key, int value) {
        if(exist[key]){
            val[key] = value;
            remove(key);
            add_to_head(key);
        }
        else{
            if(size == cap){
                int lruKey = prev[tail];
                remove(lruKey);
                exist[lruKey] = false;
                size--;
            }
            // 插入新节点
            val[key] = value;
            add_to_head(key);
            exist[key] = true;
            size++;
        }
    }
};

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache* obj = new LRUCache(capacity);
 * int param_1 = obj->get(key);
 * obj->put(key,value);
 */
 ```

### [2. Add Two Numbers](https://leetcode.com/problems/add-two-numbers/description/)

根据题目模拟就行，注意处理连续进位的条件。第一版手撕出来的代码如下：
 ```cpp

 /**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        if(l1 == nullptr) return l2;
        if(l2 == nullptr) return l1;
        ListNode* dummy = new ListNode();
        ListNode* cur = dummy;
        int carry = 0;
        while(l1 && l2){
            int sum = l1->val + l2->val + carry;
            carry = sum / 10;
            sum %= 10;
            ListNode* newNode = new ListNode(sum);
            cur->next = newNode;
            cur = cur->next;
            l1 = l1->next;
            l2 = l2->next;
        }
        if(carry == 0){
            if(l1 == nullptr) cur->next = l2;
            else cur->next = l1;
        }
        else{
            // 有进位可以新建一个节点值为1的节点然后递归下去
            ListNode* left = new ListNode(1);
            ListNode* right;
            right = (l1 == nullptr ? l2 : l1);
            cur->next = addTwoNumbers(left, right);
        }
        return dummy->next;
    }
};
```

优化一下判断条件，直接统一循环即可♻️
```cpp

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode* dummy = new ListNode(0);
        ListNode* cur = dummy;
        int carry = 0;
        while(l1 != nullptr || l2 != nullptr || carry != 0){
            int p1 = (l1 == nullptr ? 0 : l1->val);
            int p2 = (l2 == nullptr ? 0 : l2->val);
            int sum = p1 + p2 + carry;
            cur->next = new ListNode(sum % 10);
            carry = sum / 10;
            if(l1) l1 = l1->next;
            if(l2) l2 = l2->next;
            cur = cur->next;
        }
        ListNode* head = dummy->next;
        delete dummy;
        return head;
    }
};
```

```cpp
class Solution {
public:
    string addStrings(string num1, string num2) {
        int p1 = num1.size() - 1;
        int p2 = num2.size() - 1;
        int carry = 0;
        string ans = "";
        while(p1 >= 0 || p2 >= 0 || carry != 0){
            int a = (p1 >=0 ? (num1[p1] - '0') : 0);
            int b = (p2 >=0 ? (num2[p2] - '0') : 0);

            carry = sum / 10;
            sum %= 10;
            char c = sum + '0';

            p1--;
            p2--;
        }
        reverse(ans.begin(), ans.end());
```

### [200. Number of Islands](https://leetcode.com/problems/number-of-islands/description/)

可以使用DFS，也可使用BFS，标准的板子题目。我觉得用DFS更舒服。

```cpp

class Solution {
public:
    vector<vector<bool>> visited;
    int dx[4] = {0, -1, 0, 1};
    int dy[4] = {-1, 0, 1, 0};
    int m, n;
    bool isValid(int x, int y, int m, int n, vector<vector<char>>& grid){
        if(x < 0 || x >= m || y < 0 || y >= n || visited[x][y] || grid[x][y] == '0') return false;
        return true;
    }
    void dfs(int x, int y, vector<vector<char>>& grid){
        visited[x][y] = true;
        for(int i = 0; i < 4; i++){
            int nx = x + dx[i];
            int ny = y + dy[i];
            if(isValid(nx, ny, m, n, grid)) dfs(nx, ny, grid);
        }
    }

    int numIslands(vector<vector<char>>& grid) {
        m = grid.size(), n = grid[0].size();
        int cnt = 0;
        visited = vector<vector<bool>>(m, vector<bool>(n, false));
        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if(grid[i][j] == '1' && !visited[i][j]){
                    cnt++;
                    dfs(i, j, grid);
                }
            }
        }
        return cnt;
    }
};
```

### 面试原题（字符串贪心）
面试的时候，得到了这样的一道题目。
```

/*
问题描述：
给定两个整数 A 和 B，返回任意字符串 S，要求满足：
S 的长度为 A + B，且正好包含 A 个 'a' 字母与 B 个 'b' 字母；
子串 'aaa' 没有出现在 S 中；
子串 'bbb' 没有出现在 S 中。

示例 1：
输入：A = 1, B = 2
输出："abb"
解释："abb", "bab" 和 "bba" 都是正确答案。

示例 2：
输入：A = 4, B = 1
输出："aabaa"

提示：
0 <= A <= 100
0 <= B <= 100

对于给定的 A 和 B，保证存在满足要求的 S。
*/ 
```

这道题是一道经典的贪心算法（Greedy Algorithm）问题。核心思路是：每次尽可能多地消耗剩余数量较多的那个字母，除非这么做会导致出现连续三个相同字母。

```cpp

#include <string>

using namespace std;

class Solution {
public:
    string strWithout3a3b(int A, int B) {
        string result = "";
        
        while (A > 0 || B > 0) {
            int n = result.length();
            bool writeA = false;
            
            // 检查最后两个字符是否相同
            if (n >= 2 && result[n - 1] == result[n - 2]) {
                // 如果最后两个字符都是 'b'，那么这次必须写 'a'
                if (result[n - 1] == 'b') {
                    writeA = true;
                } else {
                    // 如果最后两个字符都是 'a'，那么这次必须写 'b' (writeA 保持 false)
                    writeA = false; 
                }
            } else {
                // 如果没有连续相同的限制，谁剩的多就写谁
                if (A >= B) {
                    writeA = true;
                } else {
                    writeA = false;
                }
            }
            
            // 根据决策写入字符，并扣减对应数量
            if (writeA) {
                result += 'a';
                A--;
            } else {
                result += 'b';
                B--;
            }
        }
        
        return result;
    }
};
```

### [7. Reverse Integer](https://leetcode.com/problems/reverse-integer/description/)
题目有限制：直接用long存sum值然后强制类型转换即可
```cpp

class Solution {
public:
    int reverse(int x) {
        long long sum = 0; // 使用 64 位整数暂存结果，避免运算中溢出
        
        while (x != 0) {
            int digit = x % 10;
            sum = sum * 10 + digit;
            x /= 10;
        }
        
        // 翻转完成后，统一判断是否超出了 32 位带符号整数的范围
        if (sum > INT_MAX || sum < INT_MIN) {
            return 0;
        }
        
        // 强转回 int 返回
        return static_cast<int>(sum);
    }
};
```

### [79. Word Search](https://leetcode.com/problems/word-search/description/)
在面试的时候碰到这道题了，思路确实很简单，但就是控制DFS和回溯的时候，细节挺多的。我觉得这道题的练习意义非常重大，相当于把DFS和回溯全部练了一遍了。标准套模板的代码如下。

```cpp
class Solution {
public:
    // 偏移数组
    vector<int> dx = {0, -1, 0, 1};
    vector<int> dy = {-1, 0, 1, 0};
    // 矩阵规模
    int m, n;
    // 访问数组
    vector<vector<bool>> visited;
    // 越界判断: 出边界 or 已访问(放在边界后/防止越界)
    bool isValid(int x, int y, int m, int n){
        if(x < 0 || x>= m || y < 0 || y >= n || visited[x][y]) return false;
        return true;
    }
    // 回溯终止判断：找到路径
    bool find;

    // DFS回溯
    void dfs(int x, int y, int m, int n, int idx, vector<vector<char>>& board, string word){
        if(idx == word.size()){  // 找到立刻返回，防止越界
            find = true;
            return;
        }
        for(int i = 0; i < 4; i++){
            int nx = x + dx[i];
            int ny = y + dy[i];
            if(isValid(nx, ny, m, n) && board[nx][ny] == word[idx]){
                visited[nx][ny] = true;
                dfs(nx, ny, m, n, idx + 1, board, word);
                if(find) return;
                visited[nx][ny] = false;   // 回溯！！！！！
            }
        }
    }

    bool exist(vector<vector<char>>& board, string word) {
        // 初始化
        m = board.size(), n = board[0].size();
        find = false;
        visited = vector<vector<bool>>(m, vector<bool>(n, false));
        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if(board[i][j] == word[0]){
                    visited[i][j] = true;
                    dfs(i, j, m, n, 1, board, word);
                    visited[i][j] = false;
                }
            }
        }
        return find;
    }
};
```
精简之后的做法也很不错,直接对所有点进行一个DFS,优化掉原有的visited数组，直接进行一个原地修改
```cpp
class Solution {
private:
    bool dfs(int x, int y, int idx, vector<vector<char>>& board, const string& word){
        // 越界判断
        if(x < 0|| x >= board.size() || y < 0 || y >= board[0].size() || board[x][y] != word[idx]) return false;
        // 递归终点
        if(idx == word.size() - 1) return true;
        // 原地修改
        char tmp = board[x][y];
        board[x][y] = '#';
        bool found = dfs(x, y - 1, idx + 1, board, word) ||
                     dfs(x - 1, y, idx + 1, board, word) ||
                     dfs(x, y + 1, idx + 1, board, word) ||
                     dfs(x + 1, y, idx + 1, board, word);
        // 回溯返回
        board[x][y] = tmp;
        return found;
    }
public:
    bool exist(vector<vector<char>>& board, string word) {
        for(int i = 0; i < board.size(); i++){
            for(int j = 0; j < board[0].size(); j++){
                if(dfs(i, j, 0, board, word)) return true;
            }
        }
        return false;
    }
};
```
